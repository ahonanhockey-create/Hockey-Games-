// Feature 1: Schedule Filter (Home Page) const alertFilterBtn = document.getElementById("alertFilterBtn"); const filterOutput = document.getElementById("filterOutput"); if (alertFilterBtn) { alertFilterBtn.addEventListener("click", function() { filterOutput.textContent = "🇨🇦 Displaying Canadian matchups: Bruins vs. Leafs, Oilers vs. Knights, Canucks vs. Kraken!"; filterOutput.style.color = "#90e0ef"; }); } // Feature 2: Penalty Shootout Mini-Game Logic (Game Page) const shotButtons = document.querySelectorAll(".shot-btn"); const goalScoreDisplay = document.getElementById("goalScore"); const totalShotsDisplay = document.getElementById("totalShots"); const gameMessage = document.getElementById("gameMessage"); const resetGameBtn = document.getElementById("resetGameBtn"); let goals = 0; let shots = 0; if (shotButtons.length > 0) { shotButtons.forEach(button => { button.addEventListener("click", function() { const playerChoice = button.getAttribute("data-spot"); // Goalie AI picks a random spot: left, center, or right const spots = ["left", "center", "right"]; const goalieChoice = spots[Math.floor(Math.random() * spots.length)]; shots++; if (playerChoice === goalieChoice) { // Goalie blocked it! gameMessage.textContent = `🛑 SAVED! The goalie guessed ${goalieChoice} too!`; gameMessage.style.color = "#ff6b6b"; } else { // GOAL! goals++; gameMessage.textContent = `🚨 GOAL!! You shot ${playerChoice} and the goalie went ${goalieChoice}!`; gameMessage.style.color = "#00b4d8"; } // Update scoreboard goalScoreDisplay.textContent = goals; totalShotsDisplay.textContent = shots; }); }); // Reset Game Button resetGameBtn.addEventListener("click", function() { goals = 0; shots = 0; goalScoreDisplay.textContent = goals; totalShotsDisplay.textContent = shots; gameMessage.textContent = "Game reset! Choose where to shoot!"; gameMessage.style.color = "#e0fbfc"; }); }
